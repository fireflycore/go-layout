# 项目目录

```markdown
├── .github/
│   └── dependabot.yml          # GitHub 依赖管理机器人配置
├── .run/                       # Goland 运行配置文件，用于启动项目和调试
├── cmd/
│   ├── server/
│   │   ├── app.go              # 服务实例
│   │   ├── main.go             # 服务入口
│   │   ├── wire.go             # 依赖注入入口
│   │   └── wire_gen.go         # Wire 自动生成的依赖注入代码
├── conf/
│   └── bootstrap.json          # 服务引导配置
├── depend/
│   ├── dto/                # 通过 goverter 生成的转换方法 (DTO ↔ [DO | PO])
│   │   └── user.go
│   └── protobuf/               # 通过 buf-cli 生成的 DTO 代码
│       └── user/
│           └── v1/
│               ├── user.pb.go
│               └── user_grpc.pb.go
├── internal/
│   ├── biz/                    # 业务逻辑层
│   │   ├── convert/            # 定义 DTO ↔ [DO | PO] 转换接口
│   │   │   └── user.go         # UserConvert interface
│   │   ├── model/              # 定义业务领域模型 (DO)
│   │   │   └── user.go         # User 领域对象
│   │   ├── repo/               # 定义业务领域依赖的数据接口
│   │   │   ├── user.go         # UserRepo interface
│   │   │   └── transaction.go  # TransactionRepo interface, 用于跨业务事物的组装
│   │   └── user.go             # UserUseCase 实现
│   ├── conf/                   # 配置管理
│   │   ├── bootstrap.go        # 定义引导配置，服务必要的配置
│   │   ├── core.go             # 配置层入口
│   │   ├── conf.go             # 定义配置加载器，用于聚合配置
│   │   ├── etcd.go             # 定义获取 etcd 配置的方法，实现 Loader 接口
│   │   ├── mysql.go            # 定义获取 mysql 配置的方法，实现 Loader 接口
│   │   ├── loader.go           # 定义获取配置的接口，包含 Local/Remote 的方法
│   │   └── utils.go            # 配置辅助方法，其中定义了远程/本地配置加载方法
│   ├── data/                   # 数据访问层
│   │   ├── entity/             # 定义数据模型 (PO)
│   │   │   └── user.go         # UserPO 持久化对象
│   │   ├── user.go             # UserRepo 实现 (DAO)
│   │   ├── core.go             # 数据层入口
│   │   └── data.go             # 数据库连接
│   ├── depend/                 # 基础设施依赖
│   │   ├── core.go             # 依赖层入口
│   │   └── logger.go           # 定义日志记录器
│   ├── server/                 # 服务层
│   │   ├── core.go             # 服务实例入口
│   │   ├── grpc.go             # 注册 GRPC 服务
│   │   ├── register.go         # 初始化注册服务，注册服务信息
│   │   └── server.go           # 初始化 TCP 服务器
│   ├── service/                # 应用服务层
│   │   ├── user.go             # UserService 实现
│   │   ├── core.go             # 服务层入口
│   │   └── remote.go           # 定义第三方服务入口，用于实例化第三方服务客户端
├── .gitignore
├── buf.gen.yaml                # buf cli 配置文件
├── go.mod
├── go.sum
├── LICENSE
├── makefile                    # 项目构建和任务管理
├── README.md                   # 服务使用文档
└── run.sh                      # 服务运行脚本
```
