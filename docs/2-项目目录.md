# 项目目录

```markdown
├── .github/
│   └── dependabot.yml          # GitHub 依赖管理机器人配置
├── .run/                       # Goland 运行配置文件，用于启动项目和调试
├── cmd/
│   ├── server/
│   │   ├── app.go              # 服务实例
│   │   ├── main.go             # 服务入口
│   │   ├── wire.go             # 依赖注入入口
│   │   └── wire_gen.go         # Wire 自动生成的依赖注入代码
├── conf/
│   └── bootstrap.json          # 服务引导配置
├── dep/
│   ├── dto/                    # 通过 goverter 生成的转换方法 (DTO ↔ [DO | PO])
│   └── protobuf/               # 通过 buf-cli 生成的 DTO 代码
├── internal/
│   ├── biz/                    # 业务逻辑层
│   │   ├── convert/            # 定义 DTO ↔ [DO | PO] 转换接口
│   │   │   ├── user.go         # UserConvert interface
│   │   │   └── utils.go        # convert 通用方法定义
│   │   ├── model/              # 定义业务领域模型 (DO)，非必要无需定义
│   │   │   └── user.go         # User 领域对象
│   │   ├── repo/               # 定义业务领域依赖的数据接口
│   │   │   ├── user.go         # UserRepo interface
│   │   │   ├── rs_config.go    # ConfigRepo interface, 定义配置服务使用接口
│   │   │   ├── rs_logger.go    # LoggerRepo interface, 定义日志服务使用接口
│   │   │   └── transaction.go  # TransactionRepo interface, 用于跨业务事物的组装
│   │   ├── user.go             # UserUseCase 实现
│   │   └── core.go             # 定义biz层入口和依赖注入（wire）
│   ├── conf/                   # 配置管理
│   │   ├── bootstrap.go        # 定义引导配置，服务必要的配置
│   │   ├── etcd.go             # 定义获取 etcd 配置的方法
│   │   ├── mysql.go            # 定义获取 mysql 配置的方法
│   │   ├── redis.go            # 定义获取 redis 配置的方法
│   │   ├── utils.go            # 配置辅助方法，其中定义了远程/本地配置加载方法
│   │   └── core.go             # 定义conf层入口和依赖注入（wire）
│   ├── data/                   # 数据访问层
│   │   ├── entity/             # 定义数据模型 (PO)
│   │   │   └── user.go         # UserPO 持久化对象
│   │   ├── user.go             # UserRepo 实现 (DAO)
│   │   ├── rs_config.go        # 配置服务相关方法实现（biz/repo/rs_config.go）
│   │   ├── rs_logger.go        # 日志服务相关方法实现（biz/repo/rs_logger.go）
│   │   ├── transaction.go      # 事务实现（biz/repo/transaction.go）, 用于跨业务事务处理，仅适用于当前服务，如果是跨服务事务处理，需要使用其他机制，TCC / Saga 等
│   │   ├── data.go             # 数据层连接
│   │   └── core.go             # 定义data层入口和依赖注入（wire）
│   ├── dep/                    # 基础设施依赖
│   │   ├── logger.go           # 定义日志记录器
│   │   ├── remote.go           # 定义远程服务入口，用于实例化远程服务客户端
│   │   └── core.go             # 定义dep层入口和依赖注入（wire）
│   ├── dto/                    # 数据转换层
│   │   └── core.go             # 定义dto层入口和依赖注入（wire）
│   ├── server/                 # 服务层
│   │   ├── core.go             # 服务实例入口
│   │   ├── grpc.go             # 注册 GRPC 服务
│   │   ├── register.go         # 初始化注册服务，注册服务信息
│   │   └── server.go           # 初始化 TCP 服务器
│   ├── service/                # 应用服务层
│   │   ├── user.go             # UserService 实现
│   │   ├── core.go             # 服务层入口
│   │   └── remote.go           # 定义远程服务入口，用于实例化远程服务客户端
├── .gitignore
├── buf.gen.yaml                # buf cli 配置文件
├── go.mod
├── go.sum
├── LICENSE
├── makefile                    # 项目构建和任务管理
├── README.md                   # 服务使用文档
├── update.sh                   # 服务更新脚本（可选）
└── run.sh                      # 服务运行脚本
```
